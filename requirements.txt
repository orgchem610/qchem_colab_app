# 【バージョン指定の考え方】
#   Pythonの科学技術計算系ライブラリは、メジャーバージョンが上がる際に
#   後方互換性のない変更(API変更)が入ることが珍しくありません
#   (例: ipywidgets 7→8 でファイルアップロードウィジェットの返り値の型が
#    dict から tuple に変わっており、実際に common/gui_helpers.py に両対応の
#    吸収コードを書いています)。
#
#   そのため、このファイルでは「今動くコードが将来の未知のメジャー
#   バージョンで壊れるリスク」と「バグ修正版が出ても指定が古いままで
#   困るリスク」のバランスを取り、パッケージごとに以下の使い分けをしています。
#
#     - このプロジェクトのコードが具体的なAPIの形(戻り値の型・関数シグネチャ等)
#       に強く依存しているもの(pyscf, geometric)
#         → 下限だけでなく上限も指定し(例: ">=2.8,<3")、
#           次のメジャーバージョンを自動では取り込まないようにする。
#     - 使っている機能がごく基本的で、メジャーバージョンが上がっても
#       壊れる可能性が低いもの(PyYAML, py3Dmol)
#         → 下限のみ指定する。
#
#   ただし、これはあくまで**初回インストール時**の目安です。
#   一度Colab上で正常に動作した後は、`requirements-lock.txt`
#   (pip freezeで生成される完全固定リスト)を優先して使う設計になっており
#   (ColabApp.ipynbのSetup Section参照)、そちらが実際の再現性を担保します。
#   「将来動かなくなるリスク」への最終的な備えは、このrequirements.txtの
#   指定の緩さ・厳しさではなく、リリース(mainブランチ)のときに
#   requirements-lock.txtをGitにコミットして残しておくことです。
# =============================================================================

# --- 計算エンジン本体 --------------------------------------------------------
pyscf>=2.8,<3

# 構造最適化(内部座標法)。PySCFのgeomopt機能が依存する外部パッケージ
geometric>=1.0,<2

# --- 設定ファイル(yaml.safe_load等、ごく基本的な機能のみ使用) ---------------
PyYAML>=6.0

# 構造ファイルの読み込み(.xyz/.com/.gjf/.pdb)。ase.io.read()とAtomsオブジェクトの
# 基本API(get_chemical_symbols/get_positions)はごく安定しているため下限のみ指定。
# UMA活用アプリ(uma_backend)でもASEはASE+FAIRChemCalculatorの土台として必須になる
# ため、本アプリで採用しても依存関係全体としては増加にならない見込み。
ase>=3.22

# --- GUI(ipywidgets)については、ここでは指定しない ------------------------------
# Colab標準のGUI部品の表示の仕組みは ipywidgets 7.x(Colabのランタイムには 7.7.1 が
# 入っている)向けで、8.x に更新するとGUI部品が表示されなくなることがある
# (「コードスニペット」のパネルが開くだけになる)。そのため、Colab標準のものを
# そのまま使う(requirements-uma.txt と同じ方針)。common/gui_parts.py と
# common/gui_helpers.py は、7.x・8.x のどちらでも動くように書いてある。

# --- 可視化(基本的なAPIのみ使用) ----------------------------------------------
py3Dmol>=2.1
# (plotly・kaleidoは、エネルギーグラフを自前のSVGで描く方式に変えたため、v0.2.1で不要になった)

# --- GPU4PySCF について -------------------------------------------------------
# ColabのCUDAドライババージョンに強く依存するため、このファイルには含めず、
# ColabApp.ipynb のSetup Section内で個別にインストールを試みます
# (失敗しても pyscf_backend/engine.py がCPU実行に自動フォールバックします)。
